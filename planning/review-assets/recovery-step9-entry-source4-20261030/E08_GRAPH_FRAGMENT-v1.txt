def _manifest_policy_graph(binding, envelope, check):
    """Frozen E08 steps3-9 on detached, fully domain-validated builtins.

    `check` is the Core's internally selected original work-clock check.
    This private helper returns data, never a report or live authority. It
    performs no I/O, imports, encoding, coercion, clock selection or repair.
    Allocation/check/refusal exceptions propagate without replacement.
    """
    check()

    # Step3: exact duplicate/conflict precedence. Inputs and every nested
    # leaf have already passed strict builtin/shape/domain/size validation.
    def pin_duplicates(pins):
        seen = {}
        for pin in pins:
            check()
            name = pin['id']
            if name in seen:
                old = seen[name]
                need(False, 'POLICY_DUPLICATE_ID' if old == pin else 'POLICY_PIN_CONFLICT')
            seen[name] = pin
        return seen

    def id_duplicates(names):
        seen = set()
        for name in names:
            check()
            need(name not in seen, 'POLICY_DUPLICATE_ID')
            seen.add(name)

    def disjoint_groups(groups):
        seen = set()
        for group in groups:
            check()
            for name in group:
                check()
                need(name not in seen, 'POLICY_PIN_CONFLICT')
                seen.add(name)

    external = pin_duplicates(binding['external_tcb_pins'])
    prior = pin_duplicates(binding['prior_metadata_pins'])
    id_duplicates(binding['current_output_ids'])
    id_duplicates(binding['basis_ids'])
    disjoint_groups((external, prior, (binding['current_decision_id'],),
                     binding['current_output_ids'], binding['basis_ids']))

    id_duplicates(envelope['current_output_ids'])
    id_duplicates(envelope['external_anchor_ids'])
    id_duplicates(envelope['basis_ids'])
    disjoint_groups((envelope['external_anchor_ids'],
                     (envelope['current_decision_id'],),
                     envelope['current_output_ids'], envelope['basis_ids']))

    nodes = envelope['nodes']
    by_id = {}
    for node in nodes:
        check()
        name = node['id']
        if name in by_id:
            need(False, 'POLICY_DUPLICATE_ID' if by_id[name] == node else 'POLICY_PIN_CONFLICT')
        by_id[name] = node
    for node in nodes:
        check()
        id_duplicates(node['pin_dependencies'])

    # Step4: all declarations resolve, including nonbasis nodes and envelope
    # references whose later binding comparison may itself fail.
    for group in ((binding['current_decision_id'],),
                  binding['current_output_ids'], binding['basis_ids'],
                  external, prior,
                  (envelope['current_decision_id'],),
                  envelope['current_output_ids'],
                  envelope['external_anchor_ids'], envelope['basis_ids']):
        check()
        for name in group:
            check()
            need(name in by_id, 'POLICY_UNKNOWN_NODE')
    for node in nodes:
        check()
        for parent in node['pin_dependencies']:
            check()
            need(parent in by_id, 'POLICY_UNKNOWN_NODE')

    # Step5: trust comes from the distinct, selected binding, never a role
    # label supplied only by the envelope. Ordered selections remain ordered.
    external_ids = [pin['id'] for pin in binding['external_tcb_pins']]
    for field in ('mode', 'current_decision_id', 'current_output_ids', 'basis_ids'):
        check()
        need(envelope[field] == binding[field], 'POLICY_BINDING_MISMATCH')
    need(envelope['external_anchor_ids'] == external_ids, 'POLICY_BINDING_MISMATCH')

    role_ids = {role: set() for role in (
        'EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA', 'INPUT', 'CONSUMER',
        'CURRENT_DECISION', 'CURRENT_OUTPUT')}
    for node in nodes:
        check()
        role_ids[node['role']].add(node['id'])
    need(role_ids['EXTERNAL_TCB'] == set(external), 'POLICY_BINDING_MISMATCH')
    need(role_ids['PRIOR_APPROVED_METADATA'] == set(prior), 'POLICY_BINDING_MISMATCH')
    need(role_ids['CURRENT_DECISION'] == {binding['current_decision_id']},
         'POLICY_BINDING_MISMATCH')
    need(role_ids['CURRENT_OUTPUT'] == set(binding['current_output_ids']),
         'POLICY_BINDING_MISMATCH')
    for pins in (external, prior):
        check()
        for name, pin in pins.items():
            check()
            node = by_id[name]
            need(node['bytes'] == pin['bytes'] and node['sha256'] == pin['sha256'],
                 'POLICY_BINDING_MISMATCH')
    for name in binding['basis_ids']:
        check()
        need(by_id[name]['role'] in ('INPUT', 'CONSUMER'), 'POLICY_BINDING_MISMATCH')

    # Step6: iterative DFS over the entire graph. Postorder is dependency-
    # first for the A->B (A relies on B) direction and is reused below.
    colors = {}
    postorder = []
    for node in nodes:
        check()
        name = node['id']
        if colors.get(name, 0) == 2:
            continue
        colors[name] = 1
        stack = [(name, 0)]
        while stack:
            check()
            name, offset = stack[-1]
            parents = by_id[name]['pin_dependencies']
            if offset == len(parents):
                stack.pop()
                colors[name] = 2
                postorder.append(name)
                continue
            parent = parents[offset]
            stack[-1] = (name, offset + 1)
            color = colors.get(parent, 0)
            need(color != 1, 'POLICY_CYCLE')
            if color == 0:
                colors[parent] = 1
                stack.append((parent, 0))

    # Step7: prior/external root restrictions precede forbidden ancestry.
    for node in nodes:
        check()
        role = node['role']
        if role == 'EXTERNAL_TCB':
            need(not node['pin_dependencies'], 'POLICY_DEPENDENCY_ROLE')
        elif role == 'PRIOR_APPROVED_METADATA':
            for parent in node['pin_dependencies']:
                check()
                need(by_id[parent]['role'] in ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA'),
                     'POLICY_DEPENDENCY_ROLE')

    # Step8: memoized reachability is O(V+E), without a separate walk for
    # every basis item. Nonbasis INPUT/CONSUMER nodes are checked as well.
    reaches_current = {}
    for name in postorder:
        check()
        node = by_id[name]
        forbidden = node['role'] in ('CURRENT_DECISION', 'CURRENT_OUTPUT')
        for parent in node['pin_dependencies']:
            check()
            if reaches_current[parent]:
                forbidden = True
        reaches_current[name] = forbidden
    for node in nodes:
        check()
        if node['role'] in ('INPUT', 'CONSUMER'):
            need(not reaches_current[node['id']], 'POLICY_FORBIDDEN_ANCESTRY')

    allowed = {
        'EXTERNAL_TCB': (),
        'PRIOR_APPROVED_METADATA': ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA'),
        'INPUT': ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA', 'INPUT', 'CONSUMER'),
        'CONSUMER': ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA', 'INPUT', 'CONSUMER'),
        'CURRENT_DECISION': ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA', 'INPUT', 'CONSUMER'),
        'CURRENT_OUTPUT': ('EXTERNAL_TCB', 'PRIOR_APPROVED_METADATA', 'INPUT',
                           'CONSUMER', 'CURRENT_DECISION', 'CURRENT_OUTPUT'),
    }
    for node in nodes:
        check()
        for parent in node['pin_dependencies']:
            check()
            need(by_id[parent]['role'] in allowed[node['role']], 'POLICY_DEPENDENCY_ROLE')

    # Step9: even a declared nonbasis planned node blocks EXECUTION. It is
    # retained explicitly in AUTHORING; the report remains data only.
    if envelope['mode'] == 'EXECUTION':
        for node in nodes:
            check()
            need(node['state'] not in ('PLANNED_SOURCE', 'UNBUILT'), 'POLICY_UNBUILT')
    check()
    return by_id
