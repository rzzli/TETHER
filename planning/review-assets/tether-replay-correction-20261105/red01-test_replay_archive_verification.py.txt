"""Fictional archived-replay regressions; no experiment corpus or live backend."""
import copy
from dataclasses import replace
from pathlib import Path
import shutil
import tempfile
import unittest
from unittest.mock import patch

from asof.agents import ReplayBackend, Selection, Usage, critique, make_request
from asof.artifacts import verify_run, write_run
from asof.chrono import load_snapshot
from asof.contracts import Case, Category, EvidenceStatus
from asof.scorer import ScorerArtifact
from asof.serde import ContractError, canonical, decode, encode, fingerprint, read_json, sha256_file, write_json
from asof.workflow import Budget, run_workflow

ROOT = Path(__file__).resolve().parents[1]


class ReplayArchiveVerificationTests(unittest.TestCase):
    def setUp(self):
        self.temporary = tempfile.TemporaryDirectory()
        self.addCleanup(self.temporary.cleanup)
        self.root = Path(self.temporary.name)
        self.fixture = self.root / 'fixture'
        shutil.copytree(ROOT / 'src/asof/demo/v2', self.fixture)
        self.case = decode(Case, read_json(self.fixture / 'case.json'))
        self.model = decode(ScorerArtifact, read_json(self.fixture / 'scorer.json'))
        self.store = load_snapshot(self.fixture / 'snapshot.json', self.case)
        self.output = self.root / 'run'

    def produce(self, change=None, budget=None):
        if change:
            document = read_json(self.fixture / 'replay.json')
            change(document)
            write_json(self.fixture / 'replay.json', document)
        result = run_workflow(self.case, self.store, ReplayBackend(self.fixture / 'replay.json'),
                              self.model, budget or Budget())
        write_run(self.output, result, self.store, self.model)
        return result

    @staticmethod
    def chain(events):
        previous = None
        for index, event in enumerate(events):
            event.pop('sha256', None)
            event.update(sequence=index, previous_sha256=previous)
            previous = event['sha256'] = fingerprint(event)
        return events

    def alter_archive(self, change, trace_change=None):
        inputs = read_json(self.output / 'inputs.json')
        change(inputs['configuration']['backend_provenance'])
        events = [__import__('json').loads(line) for line in (self.output / 'trace.jsonl').read_bytes().splitlines()]
        next(event for event in events if event['event'] == 'plan')['details']['configuration'] = inputs['configuration']
        if trace_change:
            trace_change(events)
        payload = b''.join(canonical(event) + b'\n' for event in self.chain(events))
        (self.output / 'trace.jsonl').write_bytes(payload)
        (self.output / 'journal.jsonl').write_bytes(payload)
        write_json(self.output / 'inputs.json', inputs)
        run = read_json(self.output / 'run.json')
        run['input_fingerprint'] = fingerprint(inputs)
        write_json(self.output / 'run.json', run)
        receipt = read_json(self.output / 'receipt.json')
        receipt['input_fingerprint'] = fingerprint(inputs)
        receipt['outputs'] = {name: sha256_file(self.output / name) for name in receipt['outputs']}
        write_json(self.output / 'receipt.json', receipt)

    @staticmethod
    def first(proof):
        return proof['document']['responses']['genetics'][0]

    def test_valid_archive_needs_no_original_replay_path_or_backend_calls(self):
        self.produce()
        (self.fixture / 'replay.json').unlink()
        with patch.object(ReplayBackend, 'quote', side_effect=AssertionError('No quote')), \
             patch.object(ReplayBackend, 'select', side_effect=AssertionError('No selection')):
            check = verify_run(self.output)
        self.assertEqual(check['semantic_consistency'], 'verified')
        self.assertEqual(check['replay_binding']['request_binding'], 'source_reconstructed')

    def test_wrong_retained_request_fingerprint_rejected_after_consistent_reseal(self):
        self.produce()
        self.alter_archive(lambda p: self.first(p).update(request_sha256='f' * 64))
        with self.assertRaisesRegex(ContractError, '[Rr]eplay|request fingerprint'):
            verify_run(self.output)

    def test_matching_fabricated_dispatch_and_entry_still_need_source_request(self):
        self.produce()
        def change_trace(events):
            next(e for e in events if e['event'] == 'dispatch')['details']['request_sha256'] = 'f' * 64
        self.alter_archive(lambda p: self.first(p).update(request_sha256='f' * 64), change_trace)
        with self.assertRaisesRegex(ContractError, '[Rr]eplay|request fingerprint'):
            verify_run(self.output)

    def test_changed_recorded_selection_rejected(self):
        self.produce()
        self.alter_archive(lambda p: self.first(p)['selection'].update(observation_ids=[]))
        with self.assertRaisesRegex(ContractError, '[Rr]eplay'):
            verify_run(self.output)

    def test_changed_recorded_usage_rejected(self):
        self.produce()
        self.alter_archive(lambda p: self.first(p)['selection']['usage'].update(input_tokens=7))
        with self.assertRaisesRegex(ContractError, '[Rr]eplay'):
            verify_run(self.output)

    def test_missing_consumed_role_rejected(self):
        self.produce()
        self.alter_archive(lambda p: p['document']['responses'].pop('genetics'))
        with self.assertRaisesRegex(ContractError, '[Rr]eplay|recorded response'):
            verify_run(self.output)

    def test_malformed_document_rejected(self):
        self.produce()
        self.alter_archive(lambda p: p['document'].update(unrecognized=True))
        with self.assertRaises(ContractError):
            verify_run(self.output)

    def test_wrong_replay_schema_rejected_even_for_zero_calls(self):
        self.produce(budget=Budget(max_calls=0))
        self.alter_archive(lambda p: p['document'].update(schema_version=1))
        with self.assertRaises(ContractError):
            verify_run(self.output)

    def test_missing_provenance_document_rejected(self):
        self.produce()
        self.alter_archive(lambda p: p.pop('document'))
        with self.assertRaises(ContractError):
            verify_run(self.output)

    def test_unused_extra_role_and_entries_remain_permitted(self):
        def extras(document):
            extra = copy.deepcopy(document['responses']['genetics'][0])
            extra['request_sha256'] = 'f' * 64
            document['responses']['genetics'].append(extra)
            document['responses']['unused_role'] = [extra]
        self.produce(extras)
        self.assertEqual(verify_run(self.output)['semantic_consistency'], 'verified')

    def test_missing_role_can_legitimately_fail_quote_without_dispatch(self):
        result = self.produce(lambda d: d['responses'].pop('genetics'))
        self.assertEqual(result.status, 'partial')
        self.assertEqual(result.calls, 7)
        self.assertEqual(verify_run(self.output)['semantic_consistency'], 'verified')

    def test_bad_request_can_legitimately_fail_quote(self):
        result = self.produce(lambda d: d['responses']['genetics'][0].update(request_sha256='f' * 64))
        self.assertEqual(result.status, 'partial')
        self.assertEqual(verify_run(self.output)['semantic_consistency'], 'verified')

    def test_zero_call_budget_is_valid_with_unused_replay_responses(self):
        result = self.produce(budget=Budget(max_calls=0))
        self.assertEqual(result.calls, 0)
        self.assertEqual(result.status, 'budget_exhausted')
        self.assertEqual(verify_run(self.output)['semantic_consistency'], 'verified')

    def retry_document(self, document):
        records = tuple(r for r in self.store.query(self.case, Category.GENETICS)
                        if r.observation.status == EvidenceStatus.OBSERVED)
        selection = Selection(('not-an-approved-id',), Usage(0, 0, 0))
        feedback = critique(selection, records) + ('Selection outside the actual bounded request',)
        second = copy.deepcopy(document['responses']['genetics'][0])
        second['request_sha256'] = fingerprint(make_request(self.case, Category.GENETICS, records, feedback))
        document['responses']['genetics'][0]['selection'] = encode(selection)
        document['responses']['genetics'].append(second)

    def test_retry_uses_feedback_and_next_entry_in_order(self):
        result = self.produce(self.retry_document)
        self.assertEqual(result.status, 'completed')
        self.assertEqual(result.calls, 9)
        self.assertEqual(verify_run(self.output)['semantic_consistency'], 'verified')

    def test_reordered_retry_entries_rejected(self):
        self.produce(self.retry_document)
        self.alter_archive(lambda p: p['document']['responses']['genetics'].reverse())
        with self.assertRaises(ContractError):
            verify_run(self.output)

    def test_archived_only_verification_does_not_read_originals_or_overstate_binding(self):
        self.produce()
        with patch('asof.artifacts.admit_snapshot', side_effect=AssertionError('No source restoration')):
            check = verify_run(self.output, verify_sources=False)
        self.assertEqual(check['source_binding'], 'archived_fingerprints_only')
        self.assertEqual(check['replay_binding']['request_binding'], 'archived_dispatch_fingerprints')

    def test_archived_only_still_checks_replay_against_dispatch(self):
        self.produce()
        self.alter_archive(lambda p: self.first(p).update(request_sha256='f' * 64))
        with self.assertRaises(ContractError):
            verify_run(self.output, verify_sources=False)

    def test_invalid_retained_replay_rejected_before_new_artifact_publication(self):
        result = self.produce()
        configuration = encode(result.configuration)
        self.first(configuration['backend_provenance'])['request_sha256'] = 'f' * 64
        events = encode(result.trace)
        next(e for e in events if e['event'] == 'plan')['details']['configuration'] = configuration
        forged = replace(result, configuration=configuration, trace=tuple(self.chain(events)))
        other = self.root / 'must-not-publish'
        with self.assertRaises(ContractError):
            write_run(other, forged, self.store, self.model)
        self.assertFalse(other.exists())

    def test_legacy_v1_receipt_scope_is_unchanged(self):
        legacy = self.root / 'legacy'
        shutil.copytree(ROOT / 'tests/fixtures/legacy-v1', legacy)
        check = verify_run(legacy, verify_sources=False)
        self.assertEqual(check['semantic_consistency'], 'unavailable_legacy_v1')
        self.assertNotIn('replay_binding', check)


if __name__ == '__main__':
    unittest.main()
